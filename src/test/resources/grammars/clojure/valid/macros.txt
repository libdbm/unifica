(ns example.macros
  (:require [clojure.string :as str]))

; A macro using syntax quote, unquote, unquote-splicing and gensyms.
(defmacro with-timing [label & body]
  `(let [start# (System/nanoTime)
         result# (do ~@body)]
     (println ~label (- (System/nanoTime) start#))
     result#))

(defn ^:private shout [s]
  (str/upper-case s))

(def counter (atom 0))
(swap! counter inc)
(println @counter)
